+# replace_literal VAR TEXT FROM TO: sets VAR to TEXT with every occurrence of
+# the literal string FROM replaced by the literal string TO (FROM is never a
+# glob, so {data-bucket} and bucket names match as text). Phase 14 repair
+# G-12: it uses only quoted-pattern removal (${s%%"FROM"*}, ${s#*"FROM"}) and
+# no ${s//pat/rep} replacement part, because bash 3.2 (macOS /bin/bash) keeps
+# the double quotes of a quoted replacement (quote removal there began in bash
+# 4.3, see shopt compat42) and bash 5.2 reads an unquoted & in it as the match
+# (patsub_replacement). Same result on bash 3.2 and bash 5.
+replace_literal() {
+  local _rl_var="$1" _rl_s="$2" _rl_from="$3" _rl_to="$4" _rl_out=""
+  if [ -n "${_rl_from}" ]; then
+    while [[ "${_rl_s}" == *"${_rl_from}"* ]]; do
+      _rl_out="${_rl_out}${_rl_s%%"${_rl_from}"*}${_rl_to}"
+      _rl_s="${_rl_s#*"${_rl_from}"}"
+    done
+  fi
+  printf -v "${_rl_var}" '%s' "${_rl_out}${_rl_s}"
+}
